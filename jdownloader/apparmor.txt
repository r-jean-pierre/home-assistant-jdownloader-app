#include <tunables/global>

profile jdownloader flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  network,
  capability,
  file,
  signal,
  unix,

  deny @{PROC}/{*,**^[0-9*],sys/kernel/shm*} wkx,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/mem rwklx,
  deny @{PROC}/kmem rwklx,
  deny @{PROC}/kcore rwklx,

  deny mount,
  deny umount,
  deny capability net_raw,

  # Allow navigation through /output and restrict file access to jdownloader.
  # The first two rules match a differing character or a longer name.
  # The third rule matches names shorter than jdownloader that share its prefix.
  deny /output/ wklmx,
  deny /output/{[^j]*,j[^d]*,jd[^o]*,jdo[^w]*,jdow[^n]*,jdown[^l]*}{,/**} rwklmx,
  deny /output/{jdownl[^o]*,jdownlo[^a]*,jdownloa[^d]*,jdownload[^e]*,jdownloade[^r]*,jdownloader?*}{,/**} rwklmx,
  deny /output/{j,jd,jdo,jdow,jdown,jdownl,jdownlo,jdownloa,jdownload,jdownloade}{,/**} rwklmx,

  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/kernel/security/** rwklx,
}
